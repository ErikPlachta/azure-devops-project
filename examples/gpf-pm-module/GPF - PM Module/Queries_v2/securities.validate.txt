-- Securities Master Query - Validation
-- Returns active securities from validation environment

SELECT
    s.SecurityID,
    s.ISIN,
    s.CUSIP,
    s.Ticker,
    s.SecurityName,
    s.SecurityType,
    s.Currency,
    s.Country,
    s.Sector,
    s.Industry,
    s.MaturityDate,
    s.Status,
    p.LastPrice,
    p.PriceDate
FROM dbo.Securities_Val s
LEFT JOIN dbo.SecurityPrices_Val p ON s.SecurityID = p.SecurityID
    AND p.PriceDate = (SELECT MAX(PriceDate) FROM dbo.SecurityPrices_Val WHERE SecurityID = s.SecurityID)
WHERE s.Status = 'Active'
ORDER BY s.Ticker
